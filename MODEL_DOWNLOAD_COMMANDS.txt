# Qwen3-0.6B model download commands
# Run on the server. The uv environment is already configured.
# The model will be stored at this exact path:
export REPO=/home/t50063365/workspace/repos/MegaKernel-vLLM
export MODEL=/home/t50063365/models/Qwen3-0.6B

cd "$REPO"
source .venv/bin/activate
export PYTHONPATH="$REPO/src:${PYTHONPATH:-}"
export TOKENIZERS_PARALLELISM=false

# Use a Hugging Face mirror for model files. This is separate from the Huawei
# Cloud PyPI mirror used for Python packages.
export HF_ENDPOINT=https://hf-mirror.com
export HF_HOME=/home/t50063365/.cache/huggingface

# Confirm the Hugging Face CLI is available.
which hf || uv pip install \
  --python "$REPO/.venv/bin/python" \
  --index-url https://mirrors.huaweicloud.com/repository/pypi/simple/ \
  "huggingface-hub>=0.30"
hf --help | head -20

# The current target directory was manually created and may contain an
# unrelated model. Move it aside instead of mixing files from two models.
if [ -d "$MODEL" ]; then
  mv "$MODEL" "${MODEL}.old-$(date +%Y%m%d-%H%M%S)"
fi
mkdir -p "$MODEL"

# Download the official Qwen3-0.6B repository through the HF mirror.
HF_ENDPOINT=https://hf-mirror.com \
hf download Qwen/Qwen3-0.6B \
  --local-dir "$MODEL" \
  --max-workers 8

# Check that the downloaded directory is really Qwen3-0.6B.
MODEL="$MODEL" python - <<'PY'
import json
import os

model = os.environ["MODEL"]
path = os.path.join(model, "config.json")
with open(path, "r", encoding="utf-8") as handle:
    config = json.load(handle)

expected = {
    "model_type": "qwen3",
    "hidden_size": 1024,
    "intermediate_size": 3072,
    "num_hidden_layers": 28,
    "num_attention_heads": 16,
    "num_key_value_heads": 8,
    "vocab_size": 151936,
}
print("config:", path)
for key, expected_value in expected.items():
    actual = config.get(key)
    print(f"{key}: {actual}")
    if actual != expected_value:
        raise SystemExit(
            f"ERROR: {key}={actual!r}, expected {expected_value!r}; "
            "this is not Qwen3-0.6B"
        )
print("Qwen3-0.6B config: OK")
PY

# Verify the project can consume this model on the A100 server.
export NANOVLLM_CUDA_ARCH=sm_80
NANOVLLM_MODEL_PATH="$MODEL" \
python -m tools.verify_server_env --require-megakernel
