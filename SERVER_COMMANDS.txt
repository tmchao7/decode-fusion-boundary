#!/usr/bin/env bash
# Active mainline server commands for the task-graph Megakernel.
#
# Usage:
#   bash SERVER_COMMANDS.txt                         # correctness
#   RUN_MODE=model bash SERVER_COMMANDS.txt           # download/check model
#   RUN_MODE=verify bash SERVER_COMMANDS.txt          # environment + extension import
#   RUN_MODE=bandwidth bash SERVER_COMMANDS.txt       # measure achievable HBM bandwidth
#   RUN_MODE=build bash SERVER_COMMANDS.txt           # clean CUDA-extension rebuild
#   RUN_MODE=benchmark bash SERVER_COMMANDS.txt      # torch eager/graph/Megakernel A/B
#   RUN_MODE=taskgraph bash SERVER_COMMANDS.txt       # kernel microbench + ABI tests
#   RUN_MODE=taskgraph-baseline bash SERVER_COMMANDS.txt  # correctness + A/B + accounting
#   RUN_MODE=taskgraph-fusion bash SERVER_COMMANDS.txt    # fusion-granularity sweep
#   RUN_MODE=all bash SERVER_COMMANDS.txt             # build, correctness, benchmark
#
# Statistics: every timed mode runs 5 measured runs and reports per-run
# dispersion (stddev/CoV) plus a bootstrap 95% CI -- see docs/02 统计口径.
# docs/06/07's historical tables used runs=3 and are only comparable within
# themselves.
# The archived kernel.cu/burst line is intentionally not exposed here. See
# branch codex/archive/frozen-kernel for historical reproduction commands.

set -euo pipefail

RUN_MODE="${RUN_MODE:-correctness}"
REPO="${REPO:-$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)}"
MODEL="${NANOVLLM_MODEL_PATH:-${QWEN3_MODEL_PATH:-$HOME/models/Qwen3-0.6B}}"
export NANOVLLM_MODEL_PATH="$MODEL"
export QWEN3_MODEL_PATH="$MODEL"
export PYTHONPATH="$REPO/src:$REPO${PYTHONPATH:+:$PYTHONPATH}"
export HF_HUB_OFFLINE="${HF_HUB_OFFLINE:-1}"
export TRANSFORMERS_OFFLINE="${TRANSFORMERS_OFFLINE:-1}"
# Do NOT default the build arch: nvcc's -arch=sm_XX emits SASS for that arch only
# (no PTX), so a forced sm_80 build cannot run a launch on an sm_120 card -- it
# fails with "no kernel image is available for execution on the device". Left
# unset, build.py detects the visible device (sm_80 on A100, sm_120 on RTX 5090).
if [[ -n "${NANOVLLM_CUDA_ARCH:-}" ]]; then
  export NANOVLLM_CUDA_ARCH
  echo "CUDA arch: forced to ${NANOVLLM_CUDA_ARCH} by the environment"
else
  echo "CUDA arch: auto-detected from the visible device (override with NANOVLLM_CUDA_ARCH=sm_XX)"
fi
export TORCH_EXTENSIONS_DIR="${TORCH_EXTENSIONS_DIR:-$REPO/.torch_extensions}"

mkdir -p "$REPO/results"
cd "$REPO"

require_model() {
  if [[ ! -f "$MODEL/config.json" ]]; then
    echo "ERROR: model config not found at $MODEL" >&2
    echo "Set NANOVLLM_MODEL_PATH to a local Qwen3-0.6B directory." >&2
    exit 1
  fi
}

select_gpu() {
  if [[ -n "${CUDA_VISIBLE_DEVICES:-}" ]]; then
    return
  fi
  local gpu
  gpu="$(nvidia-smi --query-gpu=index,memory.free --format=csv,noheader,nounits | sort -t, -k2,2nr | head -1 | cut -d, -f1 | tr -d ' ')"
  if [[ -n "$gpu" ]]; then
    export CUDA_VISIBLE_DEVICES="$gpu"
    echo "Selected GPU: $CUDA_VISIBLE_DEVICES"
  fi
}

run_model() {
  mkdir -p "$(dirname "$MODEL")"
  if [[ ! -f "$MODEL/config.json" ]]; then
    command -v hf >/dev/null 2>&1 || { echo "ERROR: hf CLI is required for RUN_MODE=model" >&2; exit 1; }
    hf download Qwen/Qwen3-0.6B --local-dir "$MODEL" --max-workers 8
  fi
  MODEL="$MODEL" python - <<'PY'
import json, os
path = os.path.join(os.environ["MODEL"], "config.json")
config = json.load(open(path, encoding="utf-8"))
expected = {
    "model_type": "qwen3", "hidden_size": 1024,
    "intermediate_size": 3072, "num_hidden_layers": 28,
    "num_attention_heads": 16, "num_key_value_heads": 8,
    "vocab_size": 151936,
}
for key, value in expected.items():
    actual = config.get(key)
    print(f"{key}: {actual}")
    if actual != value:
        raise SystemExit(f"ERROR: {key}={actual!r}; expected {value!r}")
print("Qwen3-0.6B config: OK")
PY
}

run_verify() {
  require_model
  select_gpu
  python -m tools.verify_server_env --require-megakernel --model "$MODEL"
}

# The roofline denominator must be measured on the machine that produced the
# measurement: every A100 headroom number uses 1699 GB/s measured (83.3% of the
# 2039 GB/s spec), and a 5090 must not inherit that number. Run this once per
# machine and before any accounting run.
run_bandwidth() {
  select_gpu
  python -m tools.bench_memory_bandwidth --bytes 1.2e9 --iterations 20 \
    --json results/bandwidth.json 2>&1 | tee results/a100_bandwidth.log
}

run_build() {
  require_model
  # Offline gate first: a schedule that cannot execute must never reach a paid
  # session, and a cooperative deadlock hangs the device (docs/17). Non-zero exit
  # stops the chain. --sample bounds the adversarial sweep; see docs/17 §3.
  python -m tools.verify_schedule --sample 60 \
    --json results/schedule_verification.json \
    2>&1 | tee results/schedule_verification.log
  select_gpu
  rm -rf "$TORCH_EXTENSIONS_DIR/nanovllm_megakernel_C"
  python -m nanovllm.backends.megakernel.build 2>&1 | tee results/a100_build.log
}

run_verify_extension() {
  python - <<'PY'
from nanovllm.backends.megakernel.build import get_extension
get_extension()
print("task-graph extension import: OK")
PY
}

run_correctness() {
  require_model
  select_gpu
  run_build
  python -m tools.correctness_smoke "$MODEL" \
    --workloads benchmarks/workloads.json \
    --backends torch-eager,megakernel \
    --batch-sizes 1,2,4,8 \
    --prompt-lengths 32,128 \
    --max-tokens 16 \
    --require-megakernel \
    --allow-cascade \
    --json results/a100_correctness_megakernel.json \
    2>&1 | tee results/a100_correctness_megakernel.log
}

run_benchmark() {
  require_model
  select_gpu
  run_build
  python -m benchmarks.compare "$MODEL" \
    --suite performance \
    --workloads benchmarks/workloads.json \
    --backends torch-eager,torch-cudagraph,megakernel \
    --grid \
    --grid-lengths 32,512,1920 \
    --batch-sizes 1,2,4,8 \
    --max-tokens 128 \
    --warmup 2 \
    --runs 5 \
    --baseline torch-cudagraph \
    --json results/a100_performance.json \
    --markdown results/a100_performance.md \
    2>&1 | tee results/a100_performance.log
  python -m tools.account_decode results/a100_performance.json \
    --json results/a100_account_breakdown.json \
    2>&1 | tee results/a100_account_breakdown.md
}

run_taskgraph() {
  require_model
  select_gpu
  run_build
  PYTHONPATH="$PYTHONPATH" python -m unittest \
    tests.test_taskgraph_abi tests.test_taskgraph_backend \
    tests.test_taskgraph_kernel_source -v \
    2>&1 | tee results/taskgraph_abi_backend.log
  # GPU correctness modules. The first four are unittest classes; the last three
  # are scripts -- invoking a script through unittest reports 0 tests and exits 0,
  # which is worse than not running it at all. tests/test_server_commands.py locks
  # both the coverage and the runner style.
  python -m unittest \
    tests.test_taskgraph_smoke tests.test_taskgraph_positions \
    tests.test_taskgraph_bs2 tests.test_taskgraph_multi_sm -v \
    2>&1 | tee results/taskgraph_gpu_tests.log
  python tests/test_tg_weight_check.py 2>&1 | tee results/tg_weight_check.log
  python tests/test_tg_layer_sweep.py 2>&1 | tee results/tg_layer_sweep.log
  python tests/test_tg_logits_compare.py 2>&1 | tee results/tg_logits_compare.log

  python -m tools.bench_taskgraph \
    --model "$MODEL" \
    --batch-size 1,2,4,8 \
    --context-len 1,128,512,2048 \
    --num-steps 50 \
    --warmup 5 \
    --json results/taskgraph_kernel.json \
    2>&1 | tee results/taskgraph_kernel.log
}

run_taskgraph_baseline() {
  require_model
  select_gpu
  run_build

  python -m tools.correctness_smoke "$MODEL" \
    --workloads benchmarks/workloads.json \
    --backends torch-eager,megakernel \
    --batch-sizes 1,2,4,8 \
    --max-tokens 16 \
    --require-megakernel \
    --allow-cascade \
    --json results/taskgraph_baseline_correctness.json \
    2>&1 | tee results/taskgraph_baseline_correctness.log

  # attn_splits=2 writes unnormalized partials that the combine wave renormalizes
  # (docs/11 §11.3). The split changes only the bf16 reduction order, so this is
  # the numeric comparison that the CPU-side verifier cannot make.
  python -m tools.correctness_smoke "$MODEL" \
    --workloads benchmarks/workloads.json \
    --backends torch-eager,megakernel \
    --batch-sizes 1,2,4,8 \
    --max-tokens 16 \
    --attn-splits 2 \
    --require-megakernel \
    --allow-cascade \
    --json results/taskgraph_attn_correctness.json \
    2>&1 | tee results/taskgraph_attn_correctness.log

  python -m benchmarks.compare "$MODEL" \
    --suite performance \
    --workloads benchmarks/workloads.json \
    --backends torch-eager,torch-cudagraph,megakernel \
    --grid \
    --grid-lengths 32,512,1920 \
    --batch-sizes 1,2,4,8 \
    --max-tokens 128 \
    --warmup 2 \
    --runs 5 \
    --baseline torch-cudagraph \
    --json results/taskgraph_baseline.json \
    --markdown results/taskgraph_baseline.md \
    2>&1 | tee results/taskgraph_baseline.log

  python -m tools.account_decode results/taskgraph_baseline.json \
    --json results/taskgraph_baseline_breakdown.json \
    2>&1 | tee results/taskgraph_baseline_breakdown.md
}

run_taskgraph_fusion() {
  require_model
  select_gpu
  run_build
  # Same (batch, context) grid as the main table, megakernel arm only, one row
  # per fusion granularity. FUSION_GROUPS overrides the sweep points.
  python -m tools.bench_fusion_granularity \
    --model "$MODEL" \
    --fusion-groups "${FUSION_GROUPS:-1,2,4,7,14,28}" \
    --batch-sizes 1,2,4,8 \
    --prompt-lengths 32,512,1920 \
    --max-tokens 128 \
    --warmup 2 \
    --runs 5 \
    --json results/fusion_granularity.json \
    2>&1 | tee results/fusion_granularity.log
}

case "$RUN_MODE" in
  model) run_model ;;
  verify) run_verify ;;
  bandwidth) run_bandwidth ;;
  build) run_build ;;
  correctness) run_correctness ;;
  benchmark) run_benchmark ;;
  taskgraph) run_taskgraph ;;
  taskgraph-baseline) run_taskgraph_baseline ;;
  taskgraph-fusion) run_taskgraph_fusion ;;
  all) run_build; run_correctness; run_benchmark ;;
  *) echo "Unknown RUN_MODE=$RUN_MODE" >&2; exit 2 ;;
esac
