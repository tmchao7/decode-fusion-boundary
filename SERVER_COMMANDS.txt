# Megakernel-vLLM A100 server commands
# Run these commands on the server, one section at a time.
# Current server paths:
export REPO=/home/t50063365/workspace/repos/MegaKernel-vLLM
export MODEL=/home/t50063365/models/Qwen3-0.6B

# 1) Enter the repository and fetch the latest integration branch.
cd "$REPO"
git fetch origin
git switch codex/paged-kv-megakernel-integration
git pull --ff-only
git log -1 --oneline

# 2) Check the system driver and CUDA compiler.
nvidia-smi
nvcc --version

# If nvcc is not found, stop and load the CUDA module first:
# module avail cuda
# module load cuda/<server-version>

# 3) Install uv if needed.
if ! command -v uv >/dev/null 2>&1; then
  curl -LsSf https://astral.sh/uv/install.sh | sh
  export PATH="$HOME/.local/bin:$PATH"
fi
uv --version

# 4) Configure certificates, cache behavior, timeout, and Huawei Cloud PyPI mirror.
# The mirror is used for ordinary Python packages, torch, and nvidia-* wheels.
# The official PyTorch CUDA index remains a fallback inside setup_server.sh.
export UV_SYSTEM_CERTS=true
export UV_PYTHON_DOWNLOADS=0
export UV_LINK_MODE=copy
export UV_HTTP_TIMEOUT=300
export NANOVLLM_PYPI_INDEX_URL=https://mirrors.huaweicloud.com/repository/pypi/simple/

# If the server proxy CA is not installed system-wide, uncomment this line:
# export SSL_CERT_FILE=/PATH/TO/company-ca-bundle.pem

# 5) Remove a failed old environment if this is a fresh retry.
# This only removes the local virtual environment, not the repository or model.
rm -rf .venv

# 6) Create the uv environment and install dependencies.
# This reuses existing server Python 3.10/3.11/3.12 and will not download CPython.
bash setup_server.sh
source .venv/bin/activate
export PYTHONPATH="$REPO/src:${PYTHONPATH:-}"
export TOKENIZERS_PARALLELISM=false

# 7) Verify the environment and local Qwen3-0.6B configuration.
export NANOVLLM_CUDA_ARCH=sm_80
NANOVLLM_MODEL_PATH="$MODEL" \
python -m tools.verify_server_env --require-megakernel

# 8) Compile the packaged Megakernel CUDA extension.
mkdir -p results
NANOVLLM_CUDA_ARCH=sm_80 \
python -m nanovllm.backends.megakernel.build \
  2>&1 | tee results/a100_build.log

# 9) Run Python tests before GPU correctness tests.
PYTHONPATH=src python -m unittest discover -s tests -v

# 10) Minimal single-sequence correctness test.
CUDA_LAUNCH_BLOCKING=1 \
TORCH_SHOW_CPP_STACKTRACES=1 \
NANOVLLM_CUDA_ARCH=sm_80 \
python -m tools.correctness_smoke \
  "$MODEL" \
  --prompt-lengths 8,32 \
  --batch-sizes 1 \
  --max-tokens 8 \
  --require-megakernel \
  --json results/a100_smoke_single.json \
  2>&1 | tee results/a100_smoke_single.log

# 11) Paged-KV block-boundary correctness test. Default block size is 256.
CUDA_LAUNCH_BLOCKING=1 \
TORCH_SHOW_CPP_STACKTRACES=1 \
NANOVLLM_CUDA_ARCH=sm_80 \
python -m tools.correctness_smoke \
  "$MODEL" \
  --prompt-lengths 255,256,257,512 \
  --batch-sizes 1 \
  --max-tokens 8 \
  --require-megakernel \
  --json results/a100_smoke_boundary.json \
  2>&1 | tee results/a100_smoke_boundary.log

# 12) Batch correctness test.
CUDA_LAUNCH_BLOCKING=1 \
TORCH_SHOW_CPP_STACKTRACES=1 \
NANOVLLM_CUDA_ARCH=sm_80 \
python -m tools.correctness_smoke \
  "$MODEL" \
  --prompt-lengths 32,128,255,256,257,512 \
  --batch-sizes 1,2 \
  --max-tokens 8 \
  --require-megakernel \
  --json results/a100_smoke_batch.json \
  2>&1 | tee results/a100_smoke_batch.log

# 13) Full controlled A/B benchmark. Run only after steps 10-12 pass.
NANOVLLM_CUDA_ARCH=sm_80 \
python -m benchmarks.compare \
  "$MODEL" \
  --suite performance \
  --workloads benchmarks/workloads.json \
  --backends torch-eager,torch-cudagraph,megakernel \
  --prompt-lengths 32,128,512,1024 \
  --batch-sizes 1,8,32 \
  --max-tokens 128 \
  --warmup 2 \
  --runs 10 \
  --json results/a100_performance.json \
  --markdown results/a100_performance.md \
  2>&1 | tee results/a100_performance.log

# 14) Generate plots from the benchmark JSON.
python -m tools.plot_metrics \
  results/a100_performance.json \
  --output-dir results/a100_plots

# 15) Optional short Nsight Systems profile.
NANOVLLM_CUDA_ARCH=sm_80 \
nsys profile --trace=cuda,nvtx --sample=none \
  --output=results/a100_megakernel_nsys \
  python -m benchmarks.compare \
    "$MODEL" \
    --suite performance \
    --backends megakernel \
    --prompt-lengths 32 \
    --batch-sizes 1 \
    --max-tokens 32 \
    --warmup 1 \
    --runs 1

# 16) Optional short Nsight Compute profile.
NANOVLLM_CUDA_ARCH=sm_80 \
ncu --set full --target-processes all \
  --export results/a100_megakernel_ncu \
  python -m benchmarks.compare \
    "$MODEL" \
    --suite performance \
    --backends megakernel \
    --prompt-lengths 32 \
    --batch-sizes 1 \
    --max-tokens 32 \
    --warmup 1 \
    --runs 1
