#!/usr/bin/env bash
set -euo pipefail

# Megakernel-vLLM A100 server commands
# Run this file on the server. It stops immediately when a required step fails.
# If an earlier run was suspended with Ctrl+Z, clean it first in the same shell:
# jobs -l
# fg %<job-id>
# press Ctrl+C, then verify with: nvidia-smi
# The uv environment is already configured; this file does not recreate it.

export REPO=/home/t50063365/workspace/repos/MegaKernel-vLLM
export MODEL=/home/t50063365/models/Qwen3-0.6B
cd "$REPO"

# 1) Fetch the latest integration branch.
git fetch origin
git switch codex/paged-kv-megakernel-integration
git pull --ff-only
git log -1 --oneline

# 2) Choose the least-used physical GPU from the preferred list.
# CUDA_VISIBLE_DEVICES remaps the selected physical GPU to logical GPU 0.
GPU_SELECTION=""
for candidate in 5 6 1 3; do
  free_mb="$(nvidia-smi --id="$candidate" --query-gpu=memory.free --format=csv,noheader,nounits | tr -d '[:space:]')"
  if [ -n "$free_mb" ]; then
    printf '%s %s\n' "$free_mb" "$candidate" >> /tmp/nanovllm_gpu_candidates.$$
  fi
done
if [ -f /tmp/nanovllm_gpu_candidates.$$ ]; then
  GPU_SELECTION="$(sort -nr /tmp/nanovllm_gpu_candidates.$$ | awk 'NR == 1 {print $2}')"
  rm -f /tmp/nanovllm_gpu_candidates.$$
fi
if [ -z "$GPU_SELECTION" ]; then
  echo "ERROR: none of preferred GPUs 5,6,1,3 is visible." >&2
  exit 1
fi
export CUDA_VISIBLE_DEVICES="$GPU_SELECTION"
echo "Selected physical GPU: $GPU_SELECTION (logical GPU 0 inside this process)"
nvidia-smi --id="$GPU_SELECTION"
export TORCH_EXTENSIONS_DIR="$REPO/.torch_extensions"
export NANOVLLM_CUDA_DEBUG=1

# 3) Activate the existing uv environment.
source .venv/bin/activate
export PYTHONPATH="$REPO/src:${PYTHONPATH:-}"
export TOKENIZERS_PARALLELISM=false

# 4) Use Huawei Cloud for Python packages and the system CA store.
export UV_SYSTEM_CERTS=true
export UV_PYTHON_DOWNLOADS=0
export UV_LINK_MODE=copy
export UV_HTTP_TIMEOUT=300
export NANOVLLM_PYPI_INDEX_URL=https://mirrors.huaweicloud.com/repository/pypi/simple/
# If the proxy CA is not installed system-wide, uncomment:
# export SSL_CERT_FILE=/PATH/TO/company-ca-bundle.pem

# 5) Check the A100 driver and CUDA toolkit.
nvidia-smi
nvcc --version

# 6) Download and validate the required model.
# For the complete model download procedure, use:
# cat MODEL_DOWNLOAD_COMMANDS.txt
# Then execute that file section by section.

# 7) Verify environment and Qwen3-0.6B model configuration.
export NANOVLLM_CUDA_ARCH=sm_80
NANOVLLM_MODEL_PATH="$MODEL" \
python -m tools.verify_server_env --require-megakernel

# 8) Compile the packaged Megakernel CUDA extension.
mkdir -p results
NANOVLLM_CUDA_ARCH=sm_80 \
python -m nanovllm.backends.megakernel.build \
  2>&1 | tee results/a100_build.log

# 9) Run Python tests before GPU correctness tests.
PYTHONPATH=src python -m unittest discover -s tests -v

# 10) Minimal single-sequence correctness test.
timeout 180s env CUDA_LAUNCH_BLOCKING=1 \
TORCH_SHOW_CPP_STACKTRACES=1 \
NANOVLLM_CUDA_ARCH=sm_80 \
python -m tools.correctness_smoke \
  "$MODEL" \
  --prompt-lengths 8,32 \
  --batch-sizes 1 \
  --max-tokens 8 \
  --require-megakernel \
  --json results/a100_smoke_single.json \
  2>&1 | tee results/a100_smoke_single.log

# 11) Paged-KV block-boundary correctness test. Default block size is 256.
timeout 180s env CUDA_LAUNCH_BLOCKING=1 \
TORCH_SHOW_CPP_STACKTRACES=1 \
NANOVLLM_CUDA_ARCH=sm_80 \
python -m tools.correctness_smoke \
  "$MODEL" \
  --prompt-lengths 255,256,257,512 \
  --batch-sizes 1 \
  --max-tokens 8 \
  --require-megakernel \
  --json results/a100_smoke_boundary.json \
  2>&1 | tee results/a100_smoke_boundary.log

# 12) Batch correctness test.
timeout 180s env CUDA_LAUNCH_BLOCKING=1 \
TORCH_SHOW_CPP_STACKTRACES=1 \
NANOVLLM_CUDA_ARCH=sm_80 \
python -m tools.correctness_smoke \
  "$MODEL" \
  --prompt-lengths 32,128,255,256,257,512 \
  --batch-sizes 1,2 \
  --max-tokens 8 \
  --require-megakernel \
  --json results/a100_smoke_batch.json \
  2>&1 | tee results/a100_smoke_batch.log

# 13) Full controlled A/B benchmark. Run only after steps 10-12 pass.
NANOVLLM_CUDA_ARCH=sm_80 \
python -m benchmarks.compare \
  "$MODEL" \
  --suite performance \
  --workloads benchmarks/workloads.json \
  --backends torch-eager,torch-cudagraph,megakernel \
  --prompt-lengths 32,128,512,1024 \
  --batch-sizes 1,8,32 \
  --max-tokens 128 \
  --warmup 2 \
  --runs 10 \
  --json results/a100_performance.json \
  --markdown results/a100_performance.md \
  2>&1 | tee results/a100_performance.log

# 14) Generate plots.
python -m tools.plot_metrics \
  results/a100_performance.json \
  --output-dir results/a100_plots

# 15) Optional short Nsight Systems profile.
NANOVLLM_CUDA_ARCH=sm_80 \
nsys profile --trace=cuda,nvtx --sample=none \
  --output=results/a100_megakernel_nsys \
  python -m benchmarks.compare \
    "$MODEL" \
    --suite performance \
    --backends megakernel \
    --prompt-lengths 32 \
    --batch-sizes 1 \
    --max-tokens 32 \
    --warmup 1 \
    --runs 1

# 16) Optional short Nsight Compute profile.
NANOVLLM_CUDA_ARCH=sm_80 \
ncu --set full --target-processes all \
  --export results/a100_megakernel_ncu \
  python -m benchmarks.compare \
    "$MODEL" \
    --suite performance \
    --backends megakernel \
    --prompt-lengths 32 \
    --batch-sizes 1 \
    --max-tokens 32 \
    --warmup 1 \
    --runs 1
